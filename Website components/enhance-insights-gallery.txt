```tsx
gallery-animation.tsx
import React, { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';

interface GalleryItem {
  image: string;
  label: string;
  headline: string;
  href?: string;
  isVideo?: boolean;
}

interface ExpandableGalleryProps {
  items: GalleryItem[];
  className?: string;
}

const ExpandableGallery: React.FC<ExpandableGalleryProps> = ({ items, className = '' }) => {
  const [hoveredIndex, setHoveredIndex] = useState<number | null>(null);

  const getFlexValue = (index: number) => {
    if (hoveredIndex === null) return 1;
    return hoveredIndex === index ? 2 : 0.5;
  };

  return (
    <div className={className}>
      <div className="flex gap-2 h-96 w-full">
        {items.map((item, index) => (
          <motion.a
            key={index}
            href={item.href || "#"}
            className="relative cursor-pointer overflow-hidden rounded-md block"
            style={{ flex: 1 }}
            animate={{ flex: getFlexValue(index) }}
            transition={{ duration: 0.5, ease: 'easeInOut' }}
            onMouseEnter={() => setHoveredIndex(index)}
            onMouseLeave={() => setHoveredIndex(null)}
          >
            <img
              src={item.image}
              alt={item.headline}
              className="w-full h-full object-cover"
            />
            {/* readability scrim */}
            <div className="absolute inset-0 bg-gradient-to-t from-black/85 via-black/20 to-transparent" />
            {/* video play badge */}
            {item.isVideo && (
              <div className="absolute inset-0 flex items-center justify-center">
                <div className="w-14 h-14 rounded-full bg-white/90 flex items-center justify-center">
                  <svg className="w-6 h-6 text-[#124191] ml-1" fill="currentColor" viewBox="0 0 24 24">
                    <path d="M8 5v14l11-7z" />
                  </svg>
                </div>
              </div>
            )}
            {/* text overlay */}
            <div className="absolute bottom-0 left-0 right-0 p-5">
              <span className="text-xs font-semibold tracking-widest uppercase text-neutral-300">
                {item.label}
              </span>
              <h3 className="mt-2 text-white font-semibold text-lg leading-snug">
                {item.headline}
              </h3>
            </div>
          </motion.a>
        ))}
      </div>
    </div>
  );
};

export function Component() {
  const items: GalleryItem[] = [
    {
      image: "/images/insights/insight-atm.jpg",
      label: "ARTICLE",
      headline: "Why Digital Air Traffic Management Depends on the Communications Network Beneath It",
      href: "#",
    },
    {
      image: "/images/insights/insight-healthcare.jpeg",
      label: "BLOG",
      headline: "Why does AI-ready healthcare depend on deterministic, resilient, secure optical network infrastructure?",
      href: "#",
    },
    {
      image: "/images/insights/insight-aces.jpg",
      label: "VIDEO",
      headline: "Faster, safer broadband for Saudi Arabia: Inside ACES's Nokia-powered network",
      href: "#",
      isVideo: true,
    },
  ];

  return (
    <section className="w-full bg-[#0a0f1a] py-20 px-6">
      <div className="max-w-7xl mx-auto">
        <h2 className="text-3xl md:text-4xl font-bold text-white mb-10">
          Explore the latest from Nokia
        </h2>
        <ExpandableGallery items={items} className="w-full" />
        <div className="flex flex-wrap gap-8 mt-8">
          <a href="#" className="text-[#5b8dd9] hover:text-white transition-colors font-semibold">More insights →</a>
          <a href="#" className="text-[#5b8dd9] hover:text-white transition-colors font-semibold">More from our blog →</a>
          <a href="#" className="text-[#5b8dd9] hover:text-white transition-colors font-semibold">More customer successes →</a>
        </div>
      </div>
    </section>
  );
}
```
